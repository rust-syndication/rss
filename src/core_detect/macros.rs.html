2macro_rules! features {
3    (
4      @TARGET: $target:ident;
5      @MACRO_NAME: $macro_name:ident;
6      @MACRO_ATTRS: $(#[$macro_attrs:meta])*
7      $(@BIND_FEATURE_NAME: $bind_feature:tt; $feature_impl:tt; )*
8      $(@NO_RUNTIME_DETECTION: $nort_feature:tt; )*
9      $(@FEATURE: $feature:ident: $feature_lit:tt; $(#[$feature_comment:meta])*)*
10    ) => {
11        #[macro_export]
12        $(#[$macro_attrs])*
13        macro_rules! $macro_name {
14            $(
15                ($feature_lit) => {
16                    $crate::__is_feature_detected::$feature()
17                };
18            )*
19            $(
20                ($bind_feature) => { $macro_name!($feature_impl); };
21            )*
22            $(
23                ($nort_feature) => {
24                    compile_error!(
25                        concat!(
26                            stringify!(nort_feature),
27                            " feature cannot be detected at run-time"
28                        )
29                    )
30                };
31            )*
32            ($t:tt,) => {
33                    $macro_name!($t);
34            };
35            ($t:tt) => {
36                compile_error!(
37                    concat!(
38                        concat!("unknown ", stringify!($target)),
39                        concat!(" target feature: ", $t)
40                    )
41                )
42            };
43        }
44
45        /// Each variant denotes a position in a bitset for a particular feature.
46        ///
47        /// PLEASE: do not use this, it is an implementation detail subject
48        /// to change.
49        #[doc(hidden)]
50        #[allow(non_camel_case_types)]
51        #[derive(Copy, Clone)]
52        #[repr(u8)]
53        pub(crate) enum Feature {
54            $(
55                $(#[$feature_comment])*
56                $feature,
57            )*
58
59            // Do not add variants after last:
60            _last
61        }
62
63        impl Feature {
64            pub(crate) fn to_str(self) -> &'static str {
65                match self {
66                    $(Feature::$feature => $feature_lit,)*
67                    Feature::_last => unreachable!(),
68                }
69            }
70            #[cfg(feature = "std_detect_env_override")]
71            pub(crate) fn from_str(s: &str) -> Result<Feature, ()> {
72                match s {
73                    $($feature_lit => Ok(Feature::$feature),)*
74                    _ => Err(())
75                }
76            }
77        }

79        /// Each function performs run-time feature detection for a single
80        /// feature. This allow us to use stability attributes on a per feature
81        /// basis.
82        ///
83        /// PLEASE: do not use this, it is an implementation detail subject
84        /// to change.
85        #[doc(hidden)]
86        pub mod __is_feature_detected {

88
89                /// PLEASE: do not use this, it is an implementation detail
90                /// subject to change.
91                #[inline]
92                #[doc(hidden)]
93                pub fn $feature() -> bool {
94                    cfg!(target_feature = $feature_lit) ||
95                        $crate::check_for($crate::Feature::$feature)
96                }
97            )*
98        }
99    };
