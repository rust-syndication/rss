5const SINCE_1_89: bool = rustversion::cfg!(since(1.89));

9macro_rules! x86_level {
10    // Rust does not provide runtime detection for x86-64-v4's `lahfsahf`.
11    ($architecture:literal, v4) => {
12        concat!($architecture, "+avx+avx2+avx512bw+avx512cd+avx512dq+avx512f+avx512vl+bmi1+bmi2+cmpxchg16b+f16c+fma+lzcnt+movbe+popcnt+sse3+sse4.1+sse4.2+ssse3+xsave")
13    };
14    ($architecture:literal, v3) => {
15        concat!($architecture, "+avx+avx2+bmi1+bmi2+cmpxchg16b+f16c+fma+lzcnt+movbe+popcnt+sse3+sse4.1+sse4.2+ssse3+xsave")
16    };
17    ($architecture:literal, v2) => {
18        concat!($architecture, "+cmpxchg16b+popcnt+sse3+sse4.1+sse4.2+ssse3")
19    };
20}

55pub(crate) fn targets(span: Span) -> Vec<Target> {
56    TARGETS
57        .iter()
58        .filter(|target| target.1)
59        .map(|target| Target::parse(&LitStr::new(target.0, span)).unwrap())
60        .collect()
