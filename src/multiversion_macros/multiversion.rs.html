1use crate::dispatcher::{DispatchMethod, Dispatcher};
2use crate::target::Target;
3use proc_macro2::TokenStream;
4use quote::ToTokens;
5use syn::{
6    parenthesized, parse::Parser, parse_quote, punctuated::Punctuated, spanned::Spanned, token,
7    Attribute, Error, ItemFn, LitStr, Meta, ReturnType, Type,
8};
9
10pub(crate) fn make_multiversioned_fn(
11    attr: TokenStream,
12    func: ItemFn,
13) -> Result<TokenStream, syn::Error> {
14    // Modifiers such as `default` cannot be applied to the free functions generated below.
15    func.modifiers.require_empty()?;
16
17    if let ReturnType::Type(_, ty) = &func.sig.output {
18        if let Type::ImplTrait(_) = **ty {
19            return Err(Error::new(
20                ty.span(),
21                "cannot multiversion function with `impl Trait` return type",
22            ));
23        }
24    }
25
26    let mut targets: Option<Vec<Target>> = None;
27    let mut inner_attrs: Option<Vec<Attribute>> = None;
28    let mut dispatcher: Option<DispatchMethod> = None;
29
30    let parser = syn::meta::parser(|meta| {
31        if targets.is_some() && (meta.path.is_ident("targets") || meta.path.is_ident("simd")) {
32            return Err(meta.error("can't specify `targets` or `simd` multiple times"));
33        }
34
35        if meta.path.is_ident("targets") {
36            if meta.input.peek(token::Paren) {
37                let content;
38                parenthesized!(content in meta.input);
39                targets = Some(
40                    Punctuated::<Target, token::Comma>::parse_terminated(&content)?
41                        .into_iter()
42                        .collect(),
43                );
44                return Ok(());
45            } else {
46                let value = meta.value()?;
47                let s: LitStr = value.parse()?;
48
49                if s.value().as_str() == "simd" {
50                    targets = Some(crate::simd::targets(meta.path.span()));
51                    return Ok(());
52                }
53
54                return Err(meta.error("expected a list of features or \"simd\""));
55            }
56        }
57
58        if meta.path.is_ident("attrs") {
59            if inner_attrs.is_some() {
60                return Err(meta.error("can't specify `attrs` multiple times"));
61            }
62            inner_attrs = Some(Vec::new());
63            let content;
64            parenthesized!(content in meta.input);
65            inner_attrs = Some(
66                Punctuated::<Meta, token::Comma>::parse_terminated(&content)?
67                    .into_iter()
68                    .map(|meta| parse_quote! { #[#meta] })
69                    .collect(),
70            );
71            return Ok(());
72        }
73
74        if meta.path.is_ident("dispatcher") {
75            if dispatcher.is_some() {
76                return Err(meta.error("can't specify `dispatcher` multiple times"));
77            }
78            let value = meta.value()?;
79            let s: LitStr = value.parse()?;
80            dispatcher = Some(match s.value().as_str() {
81                "default" => DispatchMethod::Default,
82                "static" => DispatchMethod::Static,
83                "direct" => DispatchMethod::Direct,
84                "indirect" => DispatchMethod::Indirect,
85                _ => {
86                    return Err(meta.error("expected `default`, `static`, `direct`, or `indirect`"))
87                }
88            });
89            return Ok(());
90        };
91
92        Err(meta.error("unrecognized option"))
93    });
94
95    let span = attr.span();
96    parser.parse2(attr)?;
97
98    let targets = if let Some(targets) = targets {
99        let mut unique_targets = Vec::new();
100        for target in targets {
101            if !target.has_features_specified() {
102                // TODO add span to Target
103                return Err(Error::new(span, "target must have features specified"));
104            }
105            if !unique_targets.contains(&target) {
106                unique_targets.push(target);
107            }
108        }
109        unique_targets
110    } else {
111        return Err(Error::new(span, "expected `targets`"));
112    };
113
114    let inner_attrs = inner_attrs.unwrap_or_default();
115    let dispatcher = dispatcher.unwrap_or(DispatchMethod::Default);
116
117    Ok(Dispatcher {
118        targets,
119        func,
120        inner_attrs,
121        dispatcher,
122    }
123    .to_token_stream())
124}
